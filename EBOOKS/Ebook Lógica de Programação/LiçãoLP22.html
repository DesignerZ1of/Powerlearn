🧠 Relembrando if
let idade = 18; if (idade >= 18) { console.log("Pode entrar"); }
🧠 Relembrando else
let idade = 15; if (idade >= 18) { console.log("Pode entrar"); } else { console.log("Não pode entrar"); }
🚀 Entrando no &&
O && significa:

> E

As DUAS condições precisam ser verdadeiras.
let idade = 18; let dinheiro = 100; if (idade >= 18 && dinheiro >= 50) { console.log("Pode entrar"); }
🚀 Entrando no ||
O || significa:

> OU

Apenas UMA condição precisa ser verdadeira.
let vip = false; let idade = 20; if (vip == true || idade >= 18) { console.log("Pode entrar"); }
🚀 Misturando .includes()
let jogadores = ["Carlos", "Ana"]; if (jogadores.includes("Carlos")) { console.log("Jogador encontrado"); }
🚀 Misturando .indexOf()
let jogadores = ["Carlos", "Ana"]; let posicao = jogadores.indexOf("Ana"); console.log(posicao);
🚀 Sistema simples de verificação
let jogadores = ["Carlos", "Ana"]; let moedas = [100, 20]; function comprar(nome) { if (jogadores.includes(nome)) { let posicao = jogadores.indexOf(nome); if (moedas[posicao] >= 50) { console.log("Compra realizada"); } else { console.log("Moedas insuficientes"); } } else { console.log("Jogador não encontrado"); } }
